cmake_minimum_required(VERSION 3.24)
project(cuda_nbody LANGUAGES CXX)
option(ENABLE_CUDA "Build CUDA executable" OFF)
set(CMAKE_CXX_STANDARD 17)
add_library(common src/common.cpp src/cpu.cpp)
target_include_directories(common PUBLIC include)
add_executable(nbody_cpu src/main_cpu.cpp)
target_link_libraries(nbody_cpu PRIVATE common)
if(ENABLE_CUDA)
 include(CheckLanguage)
 check_language(CUDA)
 if(NOT CMAKE_CUDA_COMPILER)
  message(FATAL_ERROR "CUDA compiler not found")
 endif()
 enable_language(CUDA)
 find_package(CUDAToolkit REQUIRED)
 add_executable(nbody_cuda src/main_cuda.cu src/gpu.cu)
 target_include_directories(nbody_cuda PRIVATE include)
 target_link_libraries(nbody_cuda PRIVATE common CUDA::cudart)
 set_target_properties(nbody_cuda PROPERTIES CUDA_STANDARD 17)
endif()
